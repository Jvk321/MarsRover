cmake_minimum_required(VERSION 3.20)
project(MarsRover VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(raylib CONFIG REQUIRED)
include(CTest)
add_subdirectory(src)
if(BUILD_TESTING)
    function(rover_test target name)
        add_executable(${target} tests/${target}.cpp)
        target_link_libraries(${target} PRIVATE ${ARGN})
        add_test(NAME ${name} COMMAND ${target})
        set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
    endfunction()
    rover_test(astar_tests astar_dynamic_grid rover_astar)
    rover_test(simulation_tests simulation_path_reveal rover_simulation)
    rover_test(rover_tests rover_motion rover_simulation)
    rover_test(camera_tests world_camera rover_camera)
    rover_test(terrain_tests terrain_generator_editor rover_terrain rover_simulation rover_camera)
    rover_test(replanning_tests dynamic_replanning rover_simulation)
    rover_test(visual_tests visual_effects rover_visuals)
    rover_test(sandstorm_tests layered_sandstorm rover_visuals)
    rover_test(feedback_tests audio_feedback_states rover_feedback)
    rover_test(drive_audio_tests wind_engine_sonar rover_feedback)
    rover_test(release_tests presentation_release rover_feedback)

    # Device/GPU checks are explicit, not part of headless CTest.
    foreach(target audio_smoke audio_endurance visual_smoke)
        add_executable(${target} tests/${target}.cpp)
        target_link_libraries(${target} PRIVATE rover_feedback)
    endforeach()
endif()
