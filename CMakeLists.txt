cmake_minimum_required(VERSION 3.20)
project(MarsRover VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(raylib CONFIG REQUIRED)
include(CTest)
add_subdirectory(src)

    target_link_libraries(simulation_tests PRIVATE rover_simulation)
    add_test(NAME simulation_path_reveal COMMAND simulation_tests)
    add_executable(rover_tests tests/rover_tests.cpp)
    target_link_libraries(rover_tests PRIVATE rover_simulation)
    add_test(NAME rover_motion COMMAND rover_tests)
    add_executable(camera_tests tests/camera_tests.cpp)
    target_link_libraries(camera_tests PRIVATE rover_camera)
    add_test(NAME world_camera COMMAND camera_tests)
    add_executable(terrain_tests tests/terrain_tests.cpp)
    target_link_libraries(terrain_tests PRIVATE rover_terrain rover_simulation rover_camera)
    add_test(NAME terrain_generator_editor COMMAND terrain_tests)
    add_executable(replanning_tests tests/replanning_tests.cpp)
    target_link_libraries(replanning_tests PRIVATE rover_simulation)
    add_test(NAME dynamic_replanning COMMAND replanning_tests)
