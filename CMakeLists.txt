cmake_minimum_required(VERSION 3.20)
project(MarsRover VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(raylib CONFIG REQUIRED)
include(CTest)
add_subdirectory(src)
if(BUILD_TESTING)
    add_executable(astar_tests tests/astar_tests.cpp)
    target_link_libraries(astar_tests PRIVATE rover_astar)
    add_test(NAME astar_dynamic_grid COMMAND astar_tests)
    add_executable(simulation_tests tests/simulation_tests.cpp)
    target_link_libraries(simulation_tests PRIVATE rover_simulation)
    add_test(NAME simulation_path_reveal COMMAND simulation_tests)
    add_executable(rover_tests tests/rover_tests.cpp)
    target_link_libraries(rover_tests PRIVATE rover_simulation)
    add_test(NAME rover_motion COMMAND rover_tests)
endif()
